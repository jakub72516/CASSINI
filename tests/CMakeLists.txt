add_library(cassini_test_framework INTERFACE)
target_include_directories(cassini_test_framework INTERFACE ${CMAKE_CURRENT_SOURCE_DIR})

function(add_cassini_test name source)
  add_executable(${name} ${source})
  target_link_libraries(${name} PRIVATE cassini cassini_test_framework)
  add_test(NAME ${name} COMMAND ${name})
endfunction()
